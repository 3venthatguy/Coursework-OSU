points: 	.long 0xb3fe665f, 0xee4e2f1f, 0x36a54881, 0x9ecc8990, 0xa5b6a739
			.long 0x5beed572, 0x5fc2bfa7, 0x3f9fc7fb, 0xcd77d622, 0x276afa97

;-------------------------------------------------------------------------------
; Main loop here
;-------------------------------------------------------------------------------

; Add your code between the labels start and end

; Your code must satisfy all these requirements:

; 		1. 	Your code must indexed mode and label when accessing the array.
; 		2. 	Your code must work with the defined constant LENGTH, not hardcoded values.
; 		3. 	Your code must return the two elements in closest_1 and closest_2.
; 		   	Order does not matter.
; 		4.  Your code must return the absolute difference in MAD.
; 		5. 	Your code is not allowed to modify any other location in RAM.
; 			Your code is not allowed to define new variables.

start:
			; Initialize the registers
			mov.w	#0x7FFF, R4				; Initialize R4 to max signed value
			mov.w	#points, R5				; Initialize R5 to outer loop pointer address location
			mov.w	#LENGTH, R6				; Initialize R6 to values stored in LENGTH

outer_loop:
			cmp.w	#4, R6					; Check if outer loop is done (2 bytes left in set)
			jl		done					; Jump to done function if R6 has less than 2 bytes left

			; Inner loop setup
			mov.w	R5, R7					; Set R7 as inner loop pointer
			incd.w	R7						; Start from next element
			mov.w	R6, R8					; Set R8 as inner loop counter
			decd.w	R8						; Adjust for starting position

inner_loop:
			cmp.w	#2, R8					; Check if inner loop is done
			jl		next_outer				; Check if inner loop is done

			; Calculate absolute difference
			mov.w	@R5, R9					; R9 = points[outer_i]
			mov.w	@R7, R10				; R10 = points[inner_i]
			sub.w	R10, R9					; R9 = points[outer_i] - points[inner_i]

			; Get absolute value
			tst.w 	R9						; Tests if R9 is negative
			jge		abs_done				; Jumps if positive

			; Negate if negative
			inv.w	R9						; Inverts R9 from negative to positive getting the one's complement
			inc.w	R9						; Increments R9 by 1 to compensate for two-s complement

abs_done:
			; Check if this is new minimum
			cmp.w	R4, R9					; Compares with current min abs diff value
			jge		not_min					; Jump if R9 >= R4

			; Update minimum difference and closest pair
			mov.w	R9, R4
			mov.w	@R5, &closest_1			; Stores first element
			mov.w	@R7, &closest_2			; Stores second element

not_min:
			; Increment inner loop
			incd.w	R7
			decd.w	R8
			jmp		inner_loop

next_outer:
			; Increment outer loop
			incd.w	R5
			decd.w	R6
			jmp		outer_loop

done:
			; Store minimum absolute difference
			mov.w	R4, &MAD

end:		jmp		end
			nop
